## Data Structure

Each career entry contains:

```typescript
{
  // Identification
  id: string;                          // O*NET style code
  clusterId: string;                   // Career cluster
  name: string;                        // Career title
  
  // Core Information
  overview: string;                    // 1-2 sentence overview
  whatTheyDo: string;                  // Detailed responsibilities
  
  // Education Requirements
  education: {
    subjects: string[];                // Core subjects to study
    degrees: string[];                 // Relevant degrees
    certifications: string[];          // Professional certifications
    entranceExams: string[];          // Entrance exam requirements
  }
  
  // Skills & Tools
  skills: string[];                    // 8-10 core professional skills
  tools: string[];                     // Technologies and tools
  
  // Market Information
  companies: string[];                 // Top 5-8 hiring employers
  industries: string[];                // Relevant industry sectors
  
  // Demand Analysis
  currentDemand: 'high' | 'medium' | 'low';
  emergingDemand: 'high' | 'medium' | 'low';
  futureOutlook: string;              // 2-3 sentences about growth
  aiImpact: string;                   // How AI affects the role
  
  // Compensation
  salaryRange: Array<{
    min: number;
    max: number;
    currency: string;
    experience: string;               // e.g., "0-2 years", "5+ years"
    region: string;                   // Geographic region
    source: string;                   // Data source
  }>
  
  // Career Pathways
  beginner: {
    title: string;                    // Pathway title
    steps: string[];                  // Steps to get started
    duration: string;                 // Time to complete
  }
  
  advanced: {
    title: string;
    steps: string[];
    duration: string;
  }
  
  // Metadata
  tags: string[];                     // Categorization tags
  source: string;                     // Data source attribution
  createdAt: Date;                    // Creation timestamp
  updatedAt: Date;                    // Last update timestamp
}
```

